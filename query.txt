CREATE DATABASE IF NOT EXISTS db_denah_duduk;
USE db_denah_duduk;

CREATE TABLE users (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    password VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

CREATE TABLE graduation_events (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

CREATE TABLE graduation_sessions (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    graduation_event_id BIGINT NOT NULL,
    date DATE NOT NULL,
    session TINYINT NULL,
    status ENUM('draft', 'published', 'archived') NOT NULL DEFAULT 'draft',
    created_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_sessions_event FOREIGN KEY (graduation_event_id) REFERENCES graduation_events(id) ON DELETE CASCADE
);

CREATE TABLE faculties (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    code VARCHAR(255) NOT NULL UNIQUE,
    name VARCHAR(255) NOT NULL,
    color VARCHAR(255) NULL,
    created_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

CREATE TABLE study_programs (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    faculty_id BIGINT NOT NULL,
    name VARCHAR(255) NOT NULL,
    degree_level VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_study_programs_faculty FOREIGN KEY (faculty_id) REFERENCES faculties(id) ON DELETE CASCADE
);

CREATE TABLE seat_rows (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    graduation_session_id BIGINT NOT NULL,
    `row` CHAR(1) NOT NULL,
    side ENUM('left', 'right') NOT NULL,
    capacity TINYINT NOT NULL,
    created_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_seat_rows_session FOREIGN KEY (graduation_session_id) REFERENCES graduation_sessions(id) ON DELETE CASCADE
);

CREATE TABLE seats (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    seat_row_id BIGINT NOT NULL,
    local TINYINT NOT NULL,
    global SMALLINT NULL,
    category VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_seats_row FOREIGN KEY (seat_row_id) REFERENCES seat_rows(id) ON DELETE CASCADE
);

CREATE TABLE graduates (
    id BIGINT AUTO_INCREMENT PRIMARY KEY,
    graduation_session_id BIGINT NOT NULL,
    faculty_id BIGINT NOT NULL,
    study_program_id BIGINT NOT NULL,
    nrp VARCHAR(255) NOT NULL,
    name VARCHAR(255) NOT NULL,
    seat_id BIGINT NULL,
    created_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    CONSTRAINT fk_graduates_session FOREIGN KEY (graduation_session_id) REFERENCES graduation_sessions(id) ON DELETE CASCADE,
    CONSTRAINT fk_graduates_faculty FOREIGN KEY (faculty_id) REFERENCES faculties(id) ON DELETE CASCADE,
    CONSTRAINT fk_graduates_study_program FOREIGN KEY (study_program_id) REFERENCES study_programs(id) ON DELETE CASCADE,
    CONSTRAINT fk_graduates_seat FOREIGN KEY (seat_id) REFERENCES seats(id) ON DELETE CASCADE
);
